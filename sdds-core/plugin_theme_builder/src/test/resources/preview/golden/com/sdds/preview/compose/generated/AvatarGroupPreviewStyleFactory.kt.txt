internal object AvatarGroupPreviewStyleFactory : GeneratedStyleFactory<AvatarGroupStyle> {
  public override val componentName: String = "AvatarGroup"

  public override val styleQualifiedName: String = "com.sdds.compose.uikit.AvatarGroupStyle"

  public override val states: PreviewStateMap = PreviewStateMap()

  public override val unsupportedProperties: Set<String> = setOf()

  public override val gradientUnsupportedProperties: Set<String> = setOf()

  public override val iconProperties: Set<String> = setOf()

  public override val nestedStyleTypes: Map<String, String> = mapOf("avatarStyle" to
      "com.sdds.compose.uikit.AvatarStyle")

  public override val registry: Map<String, GeneratedStyleFactory<*>>
    get() = GeneratedPreviewStyleFactories.all

  @Composable
  public override fun build(
    properties: Map<String, PreviewPropertyValue>,
    theme: PreparedComposeTheme,
    context: PreviewStyleContext,
  ): AvatarGroupStyle {
    val builder = AvatarGroupStyle.builder()
    properties["avatarStyle"]?.nestedStyle<AvatarStyle>(theme, context)?.let { builder.avatarStyle(it) }
    builder.dimensions {
      properties["itemOffset"]?.dpValue(theme)?.let { itemOffset(it.base) }
      properties["itemSpacing"]?.dpValue(theme)?.let { itemSpacing(it.base) }
    }
    return builder.style()
  }

  @Composable
  public override fun create(payload: PreviewPayload, theme: PreparedComposeTheme): AvatarGroupStyle
      = build(payload.component.properties, theme,
      payload.styleContext(GeneratedPreviewStyleFactories.all))
}
