internal object BasicButtonPreviewStyleFactory : GeneratedStyleFactory<ButtonStyle> {
  public override val componentName: String = "BasicButton"

  public override val styleQualifiedName: String = "com.sdds.compose.uikit.ButtonStyle"

  public override val states: PreviewStateMap = PreviewStateMap()

  public override val unsupportedProperties: Set<String> = setOf()

  public override val gradientUnsupportedProperties: Set<String> = setOf("spinnerColor")

  public override val iconProperties: Set<String> = setOf()

  public override val nestedStyleTypes: Map<String, String> = mapOf()

  public override val registry: Map<String, GeneratedStyleFactory<*>>
    get() = GeneratedPreviewStyleFactories.all

  @Composable
  public override fun build(
    properties: Map<String, PreviewPropertyValue>,
    theme: PreparedComposeTheme,
    context: PreviewStyleContext,
  ): ButtonStyle {
    val builder = ButtonStyle.basicButtonBuilder()
    properties["shape"]?.shapeValue(theme)?.let { builder.shape(it.toStateful<CornerBasedShape>(states)) }
    properties["labelStyle"]?.typographyValue(theme)?.let { builder.labelStyle(it.toStateful<TextStyle>(states)) }
    properties["valueStyle"]?.typographyValue(theme)?.let { builder.valueStyle(it.toStateful<TextStyle>(states)) }
    properties["disableAlpha"]?.floatValue()?.let { builder.disableAlpha(it) }
    properties["loadingAlpha"]?.floatValue()?.let { builder.loadingAlpha(it) }
    builder.colors {
      properties["contentColor"]?.let { v -> v.brushValue(theme)?.let { contentColor(it.toStateful<Brush>(states)) } }
      properties["backgroundColor"]?.let { v -> v.brushValue(theme)?.let { backgroundColor(it.toStateful<Brush>(states)) } }
      properties["labelColor"]?.let { v -> v.brushValue(theme)?.let { labelColor(it.toStateful<Brush>(states)) } }
      properties["valueColor"]?.let { v -> v.brushValue(theme)?.let { valueColor(it.toStateful<Brush>(states)) } }
      properties["iconColor"]?.let { v -> v.brushValue(theme)?.let { iconColor(it.toStateful<Brush>(states)) } }
      properties["spinnerColor"]?.let { v -> if (!v.hasGradient(theme)) { v.colorValue(theme)?.let { spinnerColor(it.toInteractiveColor(states)) } } }
    }
    builder.dimensions {
      properties["height"]?.dpValue(theme)?.let { height(it.toStateful<Dp>(states)) }
      properties["paddingStart"]?.dpValue(theme)?.let { paddingStart(it.toStateful<Dp>(states)) }
      properties["paddingEnd"]?.dpValue(theme)?.let { paddingEnd(it.toStateful<Dp>(states)) }
      properties["minWidth"]?.dpValue(theme)?.let { minWidth(it.toStateful<Dp>(states)) }
      properties["iconSize"]?.dpValue(theme)?.let { iconSize(it.toStateful<Dp>(states)) }
      properties["spinnerSize"]?.dpValue(theme)?.let { spinnerSize(it.toStateful<Dp>(states)) }
      properties["spinnerStrokeWidth"]?.dpValue(theme)?.let { spinnerStrokeWidth(it.toStateful<Dp>(states)) }
      properties["iconMargin"]?.dpValue(theme)?.let { iconMargin(it.toStateful<Dp>(states)) }
      properties["valueMargin"]?.dpValue(theme)?.let { valueMargin(it.toStateful<Dp>(states)) }
    }
    return builder.style()
  }

  @Composable
  public override fun create(payload: PreviewPayload, theme: PreparedComposeTheme): ButtonStyle =
      build(payload.component.properties, theme,
      payload.styleContext(GeneratedPreviewStyleFactories.all))
}
